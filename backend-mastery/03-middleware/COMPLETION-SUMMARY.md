## Middleware Ordering Best Practice

```javascript
// Recommended order (security first, errors last)
const express = require('express');
const app = express();

// 1. Security middleware
const helmet = require('./02-helmet/middleware');
app.use(helmet.securityMiddleware);

// 2. CORS
const cors = require('./05-cors/middleware');
app.use(cors.corsMiddleware);

// 3. Compression
const compression = require('./08-compression/middleware');
app.use(compression.compressionMiddleware);

// 4. Logging  
const logging = require('./07-logging/middleware');
app.use(logging.loggingMiddleware);

// 5. Tracing
const tracing = require('./14-tracing/middleware');
app.use(tracing.tracingMiddleware);

// 6. Body parsing
const bodyParser = require('./09-body/middleware');
app.use(bodyParser.jsonParser);

// 7. Authentication
const jwt = require('./01-jwt/middleware');
app.use(jwt.jwtVerifyMiddleware);

// 8. Caching
const cache = require('./06-caching/middleware');
app.use(cache.cacheMiddleware);

// 9. Routes (before error handler)
app.get('/api/resource', (req, res) => { /* ... */ });

// 10. Error handling (ALWAYS LAST!)
const { errorHandler } = require('./03-error/middleware');
app.use(errorHandler);
```
